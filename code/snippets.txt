=== sql | Row counts for every table
Reads partition metadata, so it is instant even on huge tables.

SELECT
  s.name AS SchemaName,
  t.name AS TableName,
  SUM(p.rows) AS RowCnt
FROM sys.tables t
JOIN sys.schemas s ON s.schema_id = t.schema_id
JOIN sys.partitions p ON p.object_id = t.object_id
  AND p.index_id IN (0, 1) -- heap or clustered index only, avoids double counting
GROUP BY s.name, t.name
ORDER BY RowCnt DESC;

=== sql | Table sizes on disk
Rows plus reserved and used space per table, largest first.

SELECT
  s.name AS SchemaName,
  t.name AS TableName,
  SUM(CASE WHEN ps.index_id IN (0, 1) THEN ps.row_count ELSE 0 END) AS RowCnt,
  CAST(SUM(ps.reserved_page_count) * 8 / 1024.0 AS DECIMAL(18, 2)) AS ReservedMB,
  CAST(SUM(ps.used_page_count) * 8 / 1024.0 AS DECIMAL(18, 2)) AS UsedMB
FROM sys.dm_db_partition_stats ps
JOIN sys.tables t ON t.object_id = ps.object_id
JOIN sys.schemas s ON s.schema_id = t.schema_id
GROUP BY s.name, t.name
ORDER BY ReservedMB DESC;

=== sql | Search every table for a value
Scans all text columns in the database and lists where the value occurs. Slow on big databases. Needs SQL Server 2017 or newer.

DECLARE @search NVARCHAR(200) = N'%needle%';
DECLARE @sql NVARCHAR(MAX);

-- One counting query per text column, glued together with UNION ALL
SELECT @sql = STRING_AGG(
  CAST(
    N'SELECT ' + QUOTENAME(s.name, '''') + N' AS SchemaName, '
    + QUOTENAME(t.name, '''') + N' AS TableName, '
    + QUOTENAME(c.name, '''') + N' AS ColumnName, COUNT(*) AS Hits'
    + N' FROM ' + QUOTENAME(s.name) + N'.' + QUOTENAME(t.name)
    + N' WHERE ' + QUOTENAME(c.name) + N' LIKE @search'
    + N' HAVING COUNT(*) > 0'
  AS NVARCHAR(MAX)),
  N' UNION ALL ')
FROM sys.columns c
JOIN sys.tables t ON t.object_id = c.object_id
JOIN sys.schemas s ON s.schema_id = t.schema_id
WHERE TYPE_NAME(c.system_type_id) IN ('char', 'varchar', 'nchar', 'nvarchar');

EXEC sp_executesql @sql, N'@search NVARCHAR(200)', @search = @search;

=== sql | Foreign keys of a table
Both directions. Tables pointing at it and tables it points at.

SELECT
  fk.name AS ForeignKey,
  OBJECT_SCHEMA_NAME(fk.parent_object_id) + '.' + OBJECT_NAME(fk.parent_object_id) AS ChildTable,
  pc.name AS ChildColumn,
  OBJECT_SCHEMA_NAME(fk.referenced_object_id) + '.' + OBJECT_NAME(fk.referenced_object_id) AS ParentTable,
  rc.name AS ParentColumn,
  fk.delete_referential_action_desc AS OnDelete
FROM sys.foreign_keys fk
JOIN sys.foreign_key_columns fkc ON fkc.constraint_object_id = fk.object_id
JOIN sys.columns pc ON pc.object_id = fkc.parent_object_id
  AND pc.column_id = fkc.parent_column_id
JOIN sys.columns rc ON rc.object_id = fkc.referenced_object_id
  AND rc.column_id = fkc.referenced_column_id
WHERE fk.referenced_object_id = OBJECT_ID('dbo.tablename')
   OR fk.parent_object_id = OBJECT_ID('dbo.tablename')
ORDER BY ChildTable, fk.name, fkc.constraint_column_id;

=== sql | Indexes of a table
Key and included columns per index. Needs SQL Server 2017 or newer.

SELECT
  i.name AS IndexName,
  i.type_desc AS IndexType,
  i.is_unique AS IsUnique,
  i.is_primary_key AS IsPrimaryKey,
  STRING_AGG(CASE WHEN ic.is_included_column = 0 THEN c.name END, ', ')
    WITHIN GROUP (ORDER BY ic.key_ordinal) AS KeyColumns,
  STRING_AGG(CASE WHEN ic.is_included_column = 1 THEN c.name END, ', ')
    WITHIN GROUP (ORDER BY ic.key_ordinal) AS IncludedColumns
FROM sys.indexes i
JOIN sys.index_columns ic ON ic.object_id = i.object_id
  AND ic.index_id = i.index_id
JOIN sys.columns c ON c.object_id = ic.object_id
  AND c.column_id = ic.column_id
WHERE i.object_id = OBJECT_ID('dbo.tablename')
GROUP BY i.name, i.type_desc, i.is_unique, i.is_primary_key
ORDER BY i.is_primary_key DESC, i.name;

=== sql | Running queries and blocking
Active requests with their SQL text. A nonzero blocking_session_id names the session in the way.

SELECT
  r.session_id,
  r.blocking_session_id,
  r.status,
  r.wait_type,
  r.total_elapsed_time / 1000 AS ElapsedSec,
  DB_NAME(r.database_id) AS DatabaseName,
  s.login_name,
  s.host_name,
  t.text AS SqlText
FROM sys.dm_exec_requests r
JOIN sys.dm_exec_sessions s ON s.session_id = r.session_id
CROSS APPLY sys.dm_exec_sql_text(r.sql_handle) t
WHERE r.session_id <> @@SPID
ORDER BY r.total_elapsed_time DESC;

=== sql | Generate C# properties from a table
One property line per column, ready to paste into a class.

SELECT
  'public ' +
  CASE
    WHEN DATA_TYPE = 'bigint' THEN 'long'
    WHEN DATA_TYPE = 'int' THEN 'int'
    WHEN DATA_TYPE = 'smallint' THEN 'short'
    WHEN DATA_TYPE = 'tinyint' THEN 'byte'
    WHEN DATA_TYPE = 'bit' THEN 'bool'
    WHEN DATA_TYPE IN ('decimal', 'numeric', 'money', 'smallmoney') THEN 'decimal'
    WHEN DATA_TYPE = 'float' THEN 'double'
    WHEN DATA_TYPE = 'real' THEN 'float'
    WHEN DATA_TYPE = 'date' THEN 'DateOnly'
    WHEN DATA_TYPE = 'time' THEN 'TimeOnly'
    WHEN DATA_TYPE IN ('datetime', 'datetime2', 'smalldatetime') THEN 'DateTime'
    WHEN DATA_TYPE = 'datetimeoffset' THEN 'DateTimeOffset'
    WHEN DATA_TYPE = 'uniqueidentifier' THEN 'Guid'
    WHEN DATA_TYPE IN ('binary', 'varbinary', 'image', 'timestamp') THEN 'byte[]'
    ELSE 'string'
  END +
  CASE WHEN IS_NULLABLE = 'YES' THEN '?' ELSE '' END +
  ' ' + COLUMN_NAME + ' { get; set; }' AS Property
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'dbo'
  AND TABLE_NAME = 'tablename'
ORDER BY ORDINAL_POSITION;

  FROM dbo.tablename
)
DELETE FROM Ranked
WHERE Rn > 1;

=== sql | Upsert
Update when the row exists, insert when it does not. The lock hints stop two sessions from inserting the same key.

DECLARE @name NVARCHAR(50) = N'theme', @value NVARCHAR(100) = N'dark';

BEGIN TRANSACTION;

UPDATE dbo.Settings WITH (UPDLOCK, SERIALIZABLE)
SET Value = @value
WHERE Name = @name;

IF @@ROWCOUNT = 0
  INSERT INTO dbo.Settings (Name, Value)
  VALUES (@name, @value);

COMMIT TRANSACTION;

=== sql | Schema changes that can rerun
Guards that make a migration script safe to run twice.

-- Add a column
IF COL_LENGTH('dbo.tablename', 'NewColumn') IS NULL
  ALTER TABLE dbo.tablename ADD NewColumn NVARCHAR(100) NULL;

-- Create a table
IF OBJECT_ID('dbo.NewTable', 'U') IS NULL
  CREATE TABLE dbo.NewTable (

    Name NVARCHAR(100) NOT NULL,
    CreatedAt DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()

  SELECT 1 FROM sys.indexes
  WHERE name = 'IX_NewTable_Name' AND object_id = OBJECT_ID('dbo.NewTable')

=== cs | Dump an object as JSON
Print any object to the console while debugging. Dump returns its input, so it can sit in the middle of an expression.

using System.Text.Json;
using System.Text.Json.Serialization;

var order = new { Id = 1, Customer = new { Name = "Ada" }, Tags = new[] { "new", "paid" } };

// One line to paste anywhere
Console.WriteLine(JsonSerializer.Serialize(order, new JsonSerializerOptions { WriteIndented = true }));

// Reusable version
order.Dump();
order.Customer.Dump("customer");

public static class DumpExtensions {
    private static readonly JsonSerializerOptions Options = new() {
        WriteIndented = true,
        IncludeFields = true,                               // Tuples keep their values in fields
        ReferenceHandler = ReferenceHandler.IgnoreCycles,   // EF entities often point back at each other
        Converters = { new JsonStringEnumConverter() }
    };

    public static T Dump<T>(this T value, string? label = null) {
        if (label is not null) Console.WriteLine($"--- {label}");

        // Serializing as object prints the runtime type, not just the declared one
        Console.WriteLine(JsonSerializer.Serialize<object?>(value, Options));

        return value;
    }
}

=== cs | EF Core DbContext
Context, entities, mapping, and registration.

using Microsoft.EntityFrameworkCore;

public class AppDb(DbContextOptions<AppDb> options) : DbContext(options) {
    public DbSet<Order> Orders => Set<Order>();
    public DbSet<Customer> Customers => Set<Customer>();

    protected override void OnModelCreating(ModelBuilder model) {
        model.Entity<Order>(e => {
            e.ToTable("Orders", "dbo");
            e.Property(o => o.Total).HasPrecision(18, 2);
            e.HasIndex(o => o.CreatedAt);
            e.HasOne(o => o.Customer).WithMany(c => c.Orders).HasForeignKey(o => o.CustomerId);
        });
    }
}

public class Order {
    public int Id { get; set; }
    public int CustomerId { get; set; }
    public decimal Total { get; set; }
    public DateTime CreatedAt { get; set; }
    public Customer Customer { get; set; } = null!;
}

public class Customer {
    public int Id { get; set; }
    public string Name { get; set; } = "";
    public List<Order> Orders { get; set; } = [];
}

// Program.cs
builder.Services.AddDbContext<AppDb>(o =>
    o.UseSqlServer(builder.Configuration.GetConnectionString("Default")));

=== cs | EF Core queries
Read, load related rows, update, and bulk operations. db is an injected AppDb.

using Microsoft.EntityFrameworkCore;

var since = DateTime.UtcNow.AddDays(-30);

// Read only list with a projection
var rows = await db.Orders
    .AsNoTracking()
    .Where(o => o.CreatedAt >= since)
    .OrderByDescending(o => o.CreatedAt)
    .Select(o => new { o.Id, o.Total, Customer = o.Customer.Name })
    .Take(50)
    .ToListAsync();

// Load related rows
var order = await db.Orders
    .Include(o => o.Customer)
    .FirstOrDefaultAsync(o => o.Id == 1);

// Change tracked entities, then save once
if (order is not null) {
    order.Total = 99.50m;
    await db.SaveChangesAsync();
}

// Bulk update and delete without loading rows
await db.Orders
    .Where(o => o.CreatedAt < since)
    .ExecuteUpdateAsync(s => s.SetProperty(o => o.Total, 0m));

await db.Orders
    .Where(o => o.CreatedAt < since)
    .ExecuteDeleteAsync();

=== cs | Background service on a timer
Hosted service that does work on a fixed interval and stops cleanly.

public class Cleanup(ILogger<Cleanup> logger) : BackgroundService {
    private static readonly TimeSpan Interval = TimeSpan.FromMinutes(5);

    protected override async Task ExecuteAsync(CancellationToken stoppingToken) {
        using var timer = new PeriodicTimer(Interval);

        // Returns false when the host shuts down
        while (await timer.WaitForNextTickAsync(stoppingToken)) {
            logger.LogInformation("Cleanup ran at {Time}", DateTimeOffset.Now);
        }
    }
}

// Program.cs

=== cs | Retry with backoff
Retries a failing call and doubles the wait each time.

static async Task<T> RetryAsync<T>(Func<Task<T>> action, int maxAttempts = 3) {
    for (var attempt = 1; ; attempt++) {
        try {
            return await action();
        } catch (Exception) when (attempt < maxAttempts) {
            // Waits 1s, 2s, 4s
            await Task.Delay(TimeSpan.FromSeconds(Math.Pow(2, attempt - 1)));
        }
    }
}

using var http = new HttpClient();
var html = await RetryAsync(() => http.GetStringAsync("https://example.com"));

=== ts | Fetch JSON
GET and POST with a timeout and a real error on bad status codes.

const TIMEOUT_MS = 10_000;

async function getJson<T>(url: string): Promise<T> {
    const response = await fetch(url, { signal: AbortSignal.timeout(TIMEOUT_MS) });

    // fetch only rejects on network failure, so check the status
    if (!response.ok) throw new Error(`${response.status} ${response.statusText}`);

    return response.json() as Promise<T>;
}

async function postJson<T>(url: string, body: unknown): Promise<T> {
    const response = await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(body),
        signal: AbortSignal.timeout(TIMEOUT_MS)
    });

    if (!response.ok) throw new Error(`${response.status} ${response.statusText}`);

    return response.json() as Promise<T>;
}

=== ts | Debounce and throttle
Debounce waits for a pause. Throttle allows one call per interval.

function debounce<A extends unknown[]>(fn: (...args: A) => void, waitMs: number) {
    let timer: ReturnType<typeof setTimeout> | undefined;

    return (...args: A) => {
        clearTimeout(timer);
        timer = setTimeout(() => fn(...args), waitMs);
    };
}

function throttle<A extends unknown[]>(fn: (...args: A) => void, waitMs: number) {
    let last = 0;

    return (...args: A) => {
        const now = Date.now();
        if (now - last < waitMs) return;

        last = now;
        fn(...args);
    };
}

const onSearch = debounce((query: string) => console.log(query), 300);
const onScroll = throttle(() => console.log(window.scrollY), 100);

=== ts | Sleep and retry
Pause in async code and retry a flaky call with backoff.

const sleep = (ms: number) => new Promise<void>(resolve => setTimeout(resolve, ms));

async function retry<T>(fn: () => Promise<T>, attempts = 3, baseMs = 500): Promise<T> {
    for (let attempt = 1; ; attempt++) {
        try {
            return await fn();
        } catch (error) {
            if (attempt >= attempts) throw error;

            // Waits 500ms, 1s, 2s
            await sleep(baseMs * 2 ** (attempt - 1));
        }
    }
}

const page = await retry(() => fetch('/api/data').then(r => r.text()));

=== ts | Limit concurrent promises
Process a list with a fixed number of calls in flight, and collect partial successes.

async function mapLimit<T, R>(items: T[], limit: number, fn: (item: T) => Promise<R>): Promise<R[]> {
    const results = new Array<R>(items.length);
    let next = 0;

    // Each worker takes the next index until the list is empty
    async function worker() {
        while (next < items.length) {
            const index = next++;
            results[index] = await fn(items[index]);
        }
    }

    await Promise.all(Array.from({ length: Math.min(limit, items.length) }, worker));

    return results;
}

// Keep the successes when some calls fail
const settled = await Promise.allSettled(urls.map(url => fetch(url)));
const succeeded = settled
    .filter((r): r is PromiseFulfilledResult<Response> => r.status === 'fulfilled')
    .map(r => r.value);

type Result<T, E = Error> =
    | { ok: true; value: T }
    | { ok: false; error: E };

async function attempt<T>(promise: Promise<T>): Promise<Result<T>> {
    try {
        return { ok: true, value: await promise };
    } catch (error) {
        return { ok: false, error: error instanceof Error ? error : new Error(String(error)) };
    }
}

=== ts | Array helpers

const unique = <T>(items: T[]) => [...new Set(items)];

    if (bytes === 0) return '0 B';

    const index = Math.min(Math.floor(Math.log(bytes) / Math.log(BASE)), UNITS.length - 1);

    return `${(bytes / BASE ** index).toFixed(decimals)} ${UNITS[index]}`;
}

function shuffle<T>(items: T[]) {
    const copy = [...items];

    for (let i = copy.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [copy[i], copy[j]] = [copy[j], copy[i]];
    }

export const clickOutside: Action<HTMLElement, () => void> = (node, onOutside) => {
    function handle(event: MouseEvent) {
        if (node.contains(event.target as Node)) return;

        onOutside();

=== svelte ts | SvelteKit API endpoint
JSON route with GET and POST handlers.

// src/routes/api/orders/+server.ts
import { json } from '@sveltejs/kit';
import type { RequestHandler } from './$types';

const DEFAULT_LIMIT = 20;

export const GET: RequestHandler = async ({ url }) => {
    const limit = Number(url.searchParams.get('limit') ?? DEFAULT_LIMIT);

    return json({ limit, orders: [] });
};

export const POST: RequestHandler = async ({ request }) => {
    const order = await request.json();

    return json(order, { status: 201 });
};
